Panduan Sinkronisasi Database Supabase

Script SQL di bawah ini sudah idempotent (aman dijalankan berulang kali tanpa error)

Unduh schema.sql
Jika tabel analytics_clicks atau site_settings belum dibuat di Supabase, silakan buka Supabase SQL Editor, tempel kode di bawah, lalu klik Run. 
-- 1. Tabel Pelacakan Klik & Analitik (analytics_clicks)
CREATE TABLE IF NOT EXISTS public.analytics_clicks (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    event_type TEXT NOT NULL,
    target_name TEXT NOT NULL,
    category TEXT DEFAULT 'general',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 2. Tabel Pengaturan Toko (site_settings)
CREATE TABLE IF NOT EXISTS public.site_settings (
    key TEXT PRIMARY KEY,
    value TEXT NOT NULL,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 3. RLS Policies
ALTER TABLE public.analytics_clicks ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.site_settings ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Public can record analytics" ON public.analytics_clicks;
DROP POLICY IF EXISTS "Admin can view analytics" ON public.analytics_clicks;
DROP POLICY IF EXISTS "Public can view settings" ON public.site_settings;
DROP POLICY IF EXISTS "Admin can manage settings" ON public.site_settings;

CREATE POLICY "Public can record analytics" ON public.analytics_clicks FOR INSERT WITH CHECK (true);
CREATE POLICY "Admin can view analytics" ON public.analytics_clicks FOR SELECT TO authenticated USING (true);
CREATE POLICY "Public can view settings" ON public.site_settings FOR SELECT USING (true);
CREATE POLICY "Admin can manage settings" ON public.site_settings FOR ALL TO authenticated USING (true) WITH CHECK (true);
